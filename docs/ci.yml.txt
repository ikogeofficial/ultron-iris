# Copy this file to .github/workflows/ci.yml (GitHub web editor: Add file > Create new file).
name: CI

on:
  push:
    branches: [main]
  pull_request:

permissions:
  contents: read

jobs:
  verify:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22
      - name: Syntax check and secret scan
        run: npm run lint
      - name: Unit tests
        run: npm test
